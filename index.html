// NestJS WebSocket Gateway event pipeline:
@WebSocketGateway({ cors: true })
export class InventoryEventsGateway {
@SubscribeMessage('sync_stock_checkout')
async handleOrder(@MessageBody() payload: CartPayloadDto) {
// Emite a colas sin colapsar el front de ventas
return this.queueService.dispatchToOdoo(payload);
}
}

El servicio mantiene un caché en caliente de tasas de cambio del Banco Central y precios vigentes. Si Odoo se encuentra en un backup o proceso pesado, las cajas siguen facturando sin interrupción mediante sincronización eventual confirmada.
